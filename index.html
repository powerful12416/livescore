function getScore(m){ const st=m.status||{}; let disp=(st.display||st.short||m.minute||'').toString().trim(); const live=st.is_live===true||/^\d+'$/.test(disp)||disp.toLowerCase().includes('live')||st.elapsed>0; const fin=(st.status||'').toLowerCase()==='finished'||disp==='FT'||disp.toUpperCase()==='FT'; let hs=m.home?.score,as=m.away?.score; let valid=hs!==null&&hs!==undefined&&String(hs)!==''&&String(hs).toLowerCase()!=='null'; if(fin){ return {txt:valid?`${hs} - ${as}`:'-',time:'FT',live:false,fin:true,sched:false}; } if(live){ // Flashscore style - show minute like 48' 55' let minute=disp; if(st.elapsed) minute=st.elapsed+"'"; if(!minute.includes("'") && /^\d+$/.test(minute)) minute=minute+"'"; if(!minute || minute.toLowerCase().includes('live')) minute=(st.elapsed||'LIVE')+(st.elapsed?"'":''); return {txt:valid?`${hs} - ${as}`:'-',time:minute,sortTime:9999+parseInt(minute||0),live:true,fin:false,sched:false}; } // Not live - show France time 16:00 return {txt:'-',time:toFranceTime(m.kickoff||m.time||''),sortTime:0,live:false,fin:false,sched:true}; }
